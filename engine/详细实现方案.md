```python
def detect_trend_direction(candles_5m, candles_1h, candles_4h):
    """
    三个时间周期共振
    
    返回：
      - "strong_bull": 强牛市（三个周期都看涨）
      - "bull": 牛市（两个周期看涨）
      - "weak_bull": 弱牛市（一个周期看涨）
      - "range": 震荡
      - "weak_bear": 弱熊市
      - "bear": 熊市
      - "strong_bear": 强熊市
    """
    
    # 1. 计算各周期均线
    ma20_5m = MA(candles_5m, 20)
    ma60_5m = MA(candles_5m, 60)
    
    ma20_1h = MA(candles_1h, 20)
    ma60_1h = MA(candles_1h, 60)
    
    ma20_4h = MA(candles_4h, 20)
    ma60_4h = MA(candles_4h, 60)
    
    # 2. 判断每个周期的趋势
    trend_5m = judge_single_timeframe(price, ma20_5m, ma60_5m)
    trend_1h = judge_single_timeframe(price, ma20_1h, ma60_1h)
    trend_4h = judge_single_timeframe(price, ma20_4h, ma60_4h)
    
    # 3. 共振判断
    bull_count = count(trend_5m=="bull", trend_1h=="bull", trend_4h=="bull")
    bear_count = count(trend_5m=="bear", trend_1h=="bear", trend_4h=="bear")
    
    if bull_count >= 2:
        return "bull" (或strong_bull如果3个都看涨)
    elif bear_count >= 2:

    else:
```

##### B. 趋势强度判断
```python
def calculate_trend_strength(candles):
    """
    趋势强度打分：0-100
    
    因素：
    1. 均线排列（多头/空头排列）
    2. 均线角度（越陡越强）
    3. 价格偏离度（涨太快=可能回调）
    4. 成交量确认
    5. ATR波动率
    """
    
    score = 0
    
    # 1. 均线排列（30分）
    if price > ma20 > ma60 > ma120:
        score += 30  # 完美多头排列
    elif price > ma20 > ma60:
        score += 20
    elif price > ma20:
        score += 10
    
    # 2. 均线角度（20分）
    angle_ma20 = (ma20[-1] - ma20[-5]) / ma20[-5]  # 5期涨幅
    if angle_ma20 > 0.03:  # 涨超3%
        score += 20
    elif angle_ma20 > 0.01:
        score += 10
    
    # 3. 价格偏离度（20分，扣分项）
    deviation = (price - ma20) / ma20
    if deviation < 0.05:  # 偏离小于5%
        score += 20
    elif deviation < 0.10:
        score += 10
    # 偏离太大不加分（可能回调）
    
    # 4. 成交量确认（15分）
    volume_ratio = volume_recent / volume_ma20
    if volume_ratio > 1.2:  # 放量
        score += 15
    elif volume_ratio > 1.0:
        score += 10
    
    # 5. 波动率（15分）
    atr = calculate_atr(candles, 14)
    atr_pct = atr / price
    if 0.02 < atr_pct < 0.05:  # 适中波动
        score += 15
    elif 0.01 < atr_pct < 0.08:
        score += 10
    # 波动太小或太大都不好
    
    return score  # 0-100
```

```python
def _should_hold_position(self, pair, trend_context, tel, price):
    """
    判断是否应该继续持有，不要卖出
    
    持有条件（任一满足）：
    1. 强趋势 + 早期/中期
    2. 未达到目标利润
    3. 趋势未反转
    """
    broker = self.brokers[pair]
    
    # 1. 检查趋势
    if trend_context["direction"] in ["strong_bull", "bull"]:
        if trend_context["phase"] in ["early", "mid"]:
            # 强趋势的早中期，坚决持有
            return True
    
    # 2. 检查利润
    if broker.pos.cost > 0:
        unrealized_pnl_pct = (price - broker.pos.cost/broker.pos.base) / (broker.pos.cost/broker.pos.base)
        
        # 根据币种设置目标
        if "BTC" in pair or "ETH" in pair:
            target = 0.10  # 主流币目标10%
        else:
            target = 0.20  # 山寨币目标20%
        
        if unrealized_pnl_pct < target:
            # 还没到目标，继续持有
            return True
    
    # 3. 检查趋势反转信号
    if tel.get("r2", 0) < 0.50:  # 趋势崩溃
        return False
    
    if trend_context["direction"] == "bear":  # 转熊
        return False
    
    # 默认：趋势OK，继续持有
    return True
```

```python
class ExitManager:
    def calculate_stop_distance(self, trend_context, symbol):
        """
        根据趋势动态调整止损距离
        
        逻辑：
        - 强趋势：宽止损（避免被震出）
        - 弱趋势：紧止损（快速止损）
        - 不同币种：不同基准
        """
        
        # 1. 基础止损（根据币种）
        if "BTC" in symbol or "ETH" in symbol:
            base_stop = 0.05  # 5%
        elif "ZEC" in symbol:
            base_stop = 0.08  # 8%
        else:
            base_stop = 0.10  # 10%
        
        # 2. 根据趋势强度调整
        if trend_context:
            strength = trend_context["strength"]
            
            if strength > 80:  # 超强趋势
                multiplier = 1.5  # 止损放宽50%
            elif strength > 60:
                multiplier = 1.2
            elif strength > 40:
                multiplier = 1.0
            else:  # 弱趋势
                multiplier = 0.7  # 止损收紧30%
        else:
            multiplier = 1.0
        
        # 3. 根据趋势阶段调整
        if trend_context and trend_context["phase"] == "exhaustion":
            multiplier *= 0.6  # 衰竭期，快速止损
        
        return base_stop * multiplier
```

##### B. 分批止盈系统
```python
class ExitManager:
    def __init__(self, risk_cfg):
        # ... 原有代码 ...
        self.partial_exits = []  # 记录分批止盈
    
    def check_partial_exit(self, price, avg_cost, trend_context):
        """
        分批止盈：
        - 第一批：+20%，卖出30%
        - 第二批：+40%，卖出30%
        - 第三批：趋势反转，卖出剩余40%
        
        这样：
        - 锁定利润
        - 保留仓位吃趋势
        """
        
        if avg_cost <= 0:
            return None, 0.0
        
        profit_pct = (price - avg_cost) / avg_cost
        
        # 第一批止盈
        if profit_pct >= 0.20 and "20%" not in self.partial_exits:
            return "partial_exit_20%", 0.3  # 卖30%
        
        # 第二批止盈
        if profit_pct >= 0.40 and "40%" not in self.partial_exits:
            return "partial_exit_40%", 0.3  # 再卖30%
        
        # 第三批：趋势反转
        if trend_context and trend_context["phase"] == "exhaustion":
            if len(self.partial_exits) > 0:  # 已经有分批止盈
                return "partial_exit_reversal", 1.0  # 全卖
        
        return None, 0.0
```

### 3.2 仓位管理系统（新增）

#### 实现位置
创建新模块：`flygate/strategy/position_sizer.py`

#### 核心思路
```
当前仓位：
  - 固定10 USDT每单
  - 不考虑趋势强度

改进：
  - 趋势越强，仓位越重
  - 趋势越弱，仓位越轻
  - 不同阶段，不同仓位
```

#### 具体实现

```python
class PositionSizer:
    def calculate_position_size(self, 
                                base_size,      # 基础仓位（如10 USDT）
                                trend_context,  # 趋势信息
                                symbol,         # 币种
                                equity):        # 当前权益
        """
        动态仓位计算
        
        返回：实际仓位大小（USDT）
        """
        
        # 1. 基础仓位
        position = base_size
        
        # 2. 根据趋势强度调整
        if trend_context:
            strength = trend_context["strength"]
            
            if strength > 80:  # 超强趋势
                multiplier = 2.0  # 加倍
            elif strength > 60:
                multiplier = 1.5
            elif strength > 40:
                multiplier = 1.0
            else:
                multiplier = 0.5  # 弱趋势减半
            
            position *= multiplier
        
        # 3. 根据趋势阶段调整
        if trend_context:
            phase = trend_context["phase"]
            
            if phase == "early":
                position *= 1.2  # 早期多买20%
            elif phase == "mid":
                position *= 1.0
            elif phase == "late":
                position *= 0.7  # 后期减仓
            elif phase == "exhaustion":
                position *= 0.3  # 衰竭期极小仓位
        
        # 4. 根据币种调整风险
        if "BTC" in symbol or "ETH" in symbol:
            risk_factor = 1.0  # 主流币
        elif "ZEC" in symbol:
            risk_factor = 0.8  # 山寨币降低20%
        else:
            risk_factor = 0.5  # 小币降低50%
        
        position *= risk_factor
        
        # 5. 总仓位限制（不超过权益的30%）
        max_position = equity * 0.3
        position = min(position, max_position)
        
        # 6. 最小仓位（交易所限制）
        min_position = 5.0
        position = max(position, min_position)
        
        return position
```

```
└──────────────────────────────────────────────────┘
                        ↓
┌──────────────────────────────────────────────────┐

│    - 计算趋势强度（0-100）                       │

└──────────────────────────────────────────────────┘
                        ↓
┌──────────────────────────────────────────────────┐
│ 3. 果蝇大脑决策（FlyBrain）                     │
│    - 视觉特征提取（保留原有）                    │
│    - 接收趋势信息（新增）                        │
│    - 根据趋势调整决策阈值（新增）                │

└──────────────────────────────────────────────────┘
                        ↓
┌──────────────────────────────────────────────────┐
│ 4. 持仓管理（Engine）                            │
│    - 如果持仓中 + 趋势向上 → 抑制卖出（新增）   │
│    - 如果趋势反转 → 允许卖出                    │
└──────────────────────────────────────────────────┘
                        ↓
┌──────────────────────────────────────────────────┐
│ 5. 仓位计算（PositionSizer）                     │

└──────────────────────────────────────────────────┘
                        ↓
┌──────────────────────────────────────────────────┐
│ 6. 风控检查（RiskManager + ExitManager）        │
│    - 自适应止损（根据趋势调整）                  │
│    - 分批止盈（新增）                            │
│    - 风控限制（原有）                            │
└──────────────────────────────────────────────────┘
                        ↓
                    执行交易
```
